cmake_minimum_required(VERSION 3.23)
project(CoreFlowApplicationHints LANGUAGES CXX)
if(NOT ANDROID OR CMAKE_SYSTEM_VERSION VERSION_LESS 33)
    message(FATAL_ERROR "CoreFlow ADPF application integration requires Android API 33+")
endif()
add_library(coreflow_application_hints INTERFACE)
target_include_directories(coreflow_application_hints INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(coreflow_application_hints INTERFACE cxx_std_20)
find_library(COREFLOW_ANDROID_HINT_LIBRARY android REQUIRED)
target_link_libraries(coreflow_application_hints INTERFACE "${COREFLOW_ANDROID_HINT_LIBRARY}")
